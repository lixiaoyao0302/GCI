cmake_minimum_required(VERSION 3.16)
project(GCI LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PCL REQUIRED COMPONENTS common io features kdtree search)
find_package(OpenMP QUIET)
add_library(gci_core src/GCI.cpp)
target_include_directories(gci_core PUBLIC include ${PCL_INCLUDE_DIRS})
target_link_directories(gci_core PUBLIC ${PCL_LIBRARY_DIRS})
target_compile_options(gci_core PUBLIC ${PCL_DEFINITIONS})
target_link_libraries(gci_core PUBLIC ${PCL_LIBRARIES})
if(OpenMP_CXX_FOUND)
  target_link_libraries(gci_core PUBLIC OpenMP::OpenMP_CXX)
endif()
if(MSVC)
  target_compile_options(gci_core PUBLIC /utf-8 /bigobj)
  target_compile_definitions(gci_core PUBLIC NOMINMAX)
endif()
